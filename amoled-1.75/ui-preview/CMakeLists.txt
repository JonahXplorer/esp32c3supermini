cmake_minimum_required(VERSION 3.20)
project(amoled_ui_preview C)
set(CMAKE_C_STANDARD 11)
set(LVGL_DIR "${CMAKE_CURRENT_LIST_DIR}/../managed_components/lvgl__lvgl")
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
# Track font/render configuration even with localized MSVC include output.
set_source_files_properties(${LVGL_SOURCES} PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h")
add_library(lvgl_native STATIC ${LVGL_SOURCES})
target_include_directories(lvgl_native PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_compile_definitions(lvgl_native PUBLIC LV_CONF_INCLUDE_SIMPLE _CRT_SECURE_NO_WARNINGS)
set(SUGARAT "${CMAKE_CURRENT_LIST_DIR}/../../sugarat")
function(add_preview target)
    add_executable(${target} render.c
        "${SUGARAT}/main/ui/home_screen.c" "${SUGARAT}/main/ui/home_presenter.c" "${SUGARAT}/main/ui/ui_button.c")
    target_include_directories(${target} PRIVATE "${SUGARAT}/main" "${SUGARAT}/main/ui"
        "${SUGARAT}/components/sugarat_voice" stubs)
    target_link_libraries(${target} PRIVATE lvgl_native)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    endif()
endfunction()

add_preview(amoled_ui_preview)
target_compile_definitions(amoled_ui_preview PRIVATE SUGARAT_NATIVE_AMOLED_UI=1)
add_preview(sugarat_ui_preview)
set_target_properties(sugarat_ui_preview PROPERTIES EXCLUDE_FROM_ALL TRUE)
target_sources(sugarat_ui_preview PRIVATE sugarat_mocks.c "${SUGARAT}/main/screen.c" "${SUGARAT}/main/screen_conversations.c")
target_include_directories(sugarat_ui_preview PRIVATE "${SUGARAT}/components/sugarat_board"
    "${SUGARAT}/third_party/agent_link/include" ../../common/servo_radio)
if(MSVC)
    target_compile_options(lvgl_native PRIVATE /utf-8 /wd4244 /wd4267)
endif()
